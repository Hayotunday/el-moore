users

create table public.users (
  id uuid not null default gen_random_uuid (),
  email text not null,
  display_name text null,
  is_active boolean not null default true,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  last_login timestamp with time zone null,
  role public.role null default 'customer_service'::role,
  is_deleted boolean not null default false,
  constraint users_pkey primary key (id),
  constraint users_email_key unique (email)
) TABLESPACE pg_default;

create trigger users_set_updated_at BEFORE
update on users for EACH row
execute FUNCTION set_updated_at ();



stock_entries

create table public.stock_entries (
  id uuid not null default gen_random_uuid (),
  merchant_id uuid not null,
  product_id uuid not null,
  quantity integer not null,
  notes text null,
  status text not null default 'pending'::text,
  submitted_by uuid null,
  approved_by uuid null,
  approved_at timestamp with time zone null,
  created_at timestamp with time zone not null default timezone ('utc'::text, now()),
  updated_at timestamp with time zone not null default timezone ('utc'::text, now()),
  submitted_by_role text null,
  admin_approved uuid null,
  warehouse_approved uuid null,
  customer_service_approved uuid null,
  constraint stock_entries_pkey primary key (id),
  constraint stock_entries_approved_by_fkey foreign KEY (approved_by) references users (id) on delete set null,
  constraint stock_entries_customer_service_approved_fkey foreign KEY (customer_service_approved) references merchant_access_keys (id),
  constraint stock_entries_merchant_id_fkey foreign KEY (merchant_id) references merchants (id) on delete CASCADE,
  constraint stock_entries_product_id_fkey foreign KEY (product_id) references products (id) on delete CASCADE,
  constraint stock_entries_submitted_by_fkey foreign KEY (submitted_by) references users (id) on delete set null,
  constraint stock_entries_admin_approved_fkey foreign KEY (admin_approved) references merchant_access_keys (id),
  constraint stock_entries_warehouse_approved_fkey foreign KEY (warehouse_approved) references merchant_access_keys (id)
) TABLESPACE pg_default;

create index IF not exists idx_stock_entries_merchant_id on public.stock_entries using btree (merchant_id) TABLESPACE pg_default;

create index IF not exists idx_stock_entries_product_id on public.stock_entries using btree (product_id) TABLESPACE pg_default;

create index IF not exists idx_stock_entries_status on public.stock_entries using btree (status) TABLESPACE pg_default;

orders

create table public.orders (
  id uuid not null default gen_random_uuid (),
  customer_name text not null,
  delivery_address text null,
  phone_numbers jsonb null,
  merchant text null,
  cc_comment text null,
  items jsonb not null,
  total_amount numeric(12, 2) null default 0,
  warehouse_comment text null,
  fom_assigned uuid null,
  rider_name text null,
  landmark text null,
  payment_to_rider numeric(12, 2) null default 0,
  payment_method text null,
  payment_to_merchant numeric(12, 2) null default 0,
  payment_confirmed boolean not null default false,
  bank text null,
  fom_comment text null,
  extracted_by uuid null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  inventory_status public.delivery_status null default 'pending'::delivery_status,
  warehouse_status public.inventory_status null,
  status public.role null default 'customer_service'::role,
  fom_delivery_status public.delivery_status null default 'pending'::delivery_status,
  rider_assigned_at timestamp with time zone null,
  fom_assigned_at timestamp with time zone null,
  payment_verified_at timestamp with time zone null,
  quantity_delivered numeric null,
  amount_paid numeric null default 0.0,
  delivered_at timestamp with time zone null,
  prints numeric not null default '0'::numeric,
  constraint orders_pkey primary key (id),
  constraint orders_extracted_by_fkey foreign KEY (extracted_by) references users (id) on delete set null,
  constraint orders_fom_assigned_fkey foreign KEY (fom_assigned) references users (id) on delete set null
